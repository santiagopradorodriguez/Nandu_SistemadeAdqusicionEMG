\begin{frame}{Hardware y Blindaje: Reducción del Ruido Eléctrico}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \small
            \begin{block}{El Problema del Cable como Antena}
                Los cables comunes sin blindaje captan el zumbido de 50~Hz de la red eléctrica, generando un ruido de hasta $200\,\mu\text{V}$ que tapa las señales musculares de la cara.
            \end{block}
            \vspace{0.1cm}
            \begin{block}{Solución: Cables Mallados a Masa}
                Se armaron cables con malla metálica soldada al chasis del conector para desviar las interferencias ambientales a tierra.
            \end{block}

    \begin{columns}[c]

            \begin{block}{Fuerza Muscular Constante}
                Al decir la vocal «U», el labio genera siempre la misma fuerza ($\sim 180$--$210\,\mu\text{V}$). Lo que cambia con el cable es qué tan limpia se escucha.
            \end{block}
            \vspace{0.1cm}
            \textbf{\footnotesize Comparativa Cuantitativa en Laboratorio:}
            \vspace{0.05cm}
            \centering
            \resizebox{\textwidth}{!}{
            \begin{tabular}{lccc}
                \toprule
                \textbf{Configuración} & \textbf{Amplitud (U)} & \textbf{SNR Pico} & \textbf{Estabilidad} \\
                \midrule
                1. Sin mallar & $176 - 184\,\mu\text{V}$ & $12.5 - 15.4$ & Inestable (+260\%) \\
                2. Mallado simple & $180 - 210\,\mu\text{V}$ & $20.0 - 24.0$ & Moderada \\
                3. Mallado micrófono & $185 - 200\,\mu\text{V}$ & \textbf{25.0 -- 33.5} & \textbf{Excelente (-49\%)} \\

\begin{frame}{Sustitución Anatómica: Reemplazo del Digástrico por el Masetero}
    \begin{columns}[c]
        \begin{column}{0.47\textwidth}
            \footnotesize
            \begin{block}{Por Qué Cambiamos al Masetero}
                Abajo de la mandíbula (digástrico) los sensores se despegaban por el sudor y la gravedad. En el masetero quedan mucho más firmes y no se caen.
            \end{block}
            \vspace{0.05cm}
            \begin{block}{Firmeza y Activación Mandibular}
                Al mover la mandíbula el masetero se activa fuertemente. Junto con el orbicular, permite separar con gran claridad casi todas las vocales.
            \end{block}
            \vspace{0.05cm}
            \begin{alertblock}{Rendimiento Impresionante en 3D}
                En las pruebas 1 a 5 clasifica prácticamente todo:
                \begin{itemize}\scriptsize
                    \item \textbf{100\% de acierto en /e/ y 100\% en /i/}.
                    \item \textbf{94\% de acierto en /a/}.
                    \item Único solapamiento: /o/ y /u/ por cierre labial.
                \end{itemize}
            \end{alertblock}
        \end{column}

\begin{frame}{Dinámica Temporal: Secuencia de Disparo Masetero y Orbicular}
    \begin{columns}[c]
        \begin{column}{0.46\textwidth}
            \footnotesize
            \begin{block}{Más Allá de la Amplitud Pico}
                La fonación no solo se discrimina por cuánta fuerza hace cada músculo, sino por el \textbf{orden y retardo temporal} en que se disparan:
            \end{block}
            \vspace{0.06cm}
            \begin{itemize}
                \item \textbf{Vocal /a/ - Cierre mandibular:}\\
                El masetero se activa tarde (pico a $+118.7$\,ms tras el audio). No interviene al abrir, sino que se contrae para volver a cerrar la mandíbula.
                \vspace{0.04cm}
                \item \textbf{Vocal /i/ - Masetero anticipado:}\\
                El masetero dispara antes ($-221.0$\,ms pre-audio) fijando la mandíbula, y recién después actúa el orbicular ($+67.7$\,ms).
                \vspace{0.04cm}
                \item \textbf{Vocal /u/ - Embudo labial precoz:}\\
                El masetero queda inactivo ($2.3\,\mu\text{V}$), mientras el orbicular se anticipa fuertemente ($-239.7$\,ms) para protruir los labios.
            \end{itemize}
        \end{column}

\begin{frame}{Acondicionamiento por Reposo: Desacoplando /o/ de /u/}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \small
            \begin{block}{Impedancia Cutánea y Desvío del Reposo}
                \footnotesize La sudoración y el contacto cutáneo varían entre sesiones, desplazando la línea de base y la escala de cada sensor.
            \end{block}
            \vspace{0.06cm}
            \textbf{\footnotesize Corrección por Reposo y Rango Dinámico:}
            \begin{itemize}
                \scriptsize
                \item Tras el Supremo Tricanal ($M_{\text{supremo}}$), se calibra por canal y sesión:
                \vspace{-0.12cm}
                $$\tilde{x}_c(t) = \frac{x_c(t) - \mu_{\text{reposo}, c}}{P_{95}(x_c) - \mu_{\text{reposo}, c}}$$
                \vspace{-0.12cm}
                \item $\mu_{\text{reposo}}$: Línea de base basal previa a la fonación.
                \item $P_{95}$: Rango dinámico útil de contracción (percentil 95).
            \end{itemize}
            \vspace{0.05cm}
            \begin{alertblock}{Desacople Fundamental de /o/ y /u/}
                \scriptsize La confusión entre /o/ y /u/ no era muscular, sino deriva eléctrica. Al compensar el reposo, se separan nítidamente en el mapa.
            \end{alertblock}
        \end{column}

            \textbf{Izquierda - Sin reposo:} /o/ y /u/ colapsadas.\\

% Slide 11: El Problema de Probar en Otra Persona o Cambiar Sensores
% ------------------------------------------------------------------------------
\begin{frame}{El Problema de Probar en Otra Persona o Cambiar Sensores}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \small
            \begin{block}{La Señal Aparece Rotada o Espejada}
                Cuando evaluamos a otro participante (Santi, Petra, Candela) o volvemos a pegar los sensores otro día, el mapa interno aparece girado o invertido.
            \end{block}
            \vspace{0.1cm}
            \textbf{Por Qué Falla la Red si No la Calibramos:}
